total(extract())

report_dag()

WITH cust_total AS (
    SELECT c.name, SUM(o.amount) AS total
    FROM customers c
    JOIN orders o ON c.customer_id = o.customer_id
    WHERE o.status IS NULL OR o.status <> 'Cancelled'
    GROUP BY c.customer_id, c.name
)
SELECT name, total
FROM cust_total
WHERE total > (SELECT AVG(total) FROM cust_total)
ORDER BY total DESC;
